cmake_minimum_required(VERSION 3.22)
project(accordomi VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(AC_BUILD_JNI "Build the JNI bridge on the host" OFF)
option(AC_SANITIZE "Enable host address and undefined-behavior sanitizers" OFF)
if(AC_SANITIZE AND NOT ANDROID)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(accordomi STATIC src/dsp.c src/tuning.c src/piano.c)
set_target_properties(accordomi PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(accordomi PUBLIC include)
target_compile_options(accordomi PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(accordomi PUBLIC m)
if(ANDROID OR AC_BUILD_JNI)
    add_library(accordomi_jni SHARED src/jni.c)
    target_link_libraries(accordomi_jni PRIVATE accordomi)
    target_compile_options(accordomi_jni PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(NOT ANDROID)
        find_package(Java REQUIRED COMPONENTS Development)
        get_filename_component(AC_JAVAC "${Java_JAVAC_EXECUTABLE}" REALPATH)
        get_filename_component(AC_JAVA_BIN "${AC_JAVAC}" DIRECTORY)
        get_filename_component(AC_JAVA_HOME "${AC_JAVA_BIN}" DIRECTORY)
        find_path(AC_JNI_INCLUDE jni.h HINTS "$ENV{JAVA_HOME}/include" "${AC_JAVA_HOME}/include" REQUIRED)
        find_path(AC_JNI_PLATFORM jni_md.h HINTS "${AC_JNI_INCLUDE}/linux" "${AC_JNI_INCLUDE}/darwin" "${AC_JNI_INCLUDE}/win32" REQUIRED)
        target_include_directories(accordomi_jni PRIVATE ${AC_JNI_INCLUDE} ${AC_JNI_PLATFORM})
    endif()
endif()
if(NOT ANDROID)
    add_executable(accordomi-analyze cli/main.c cli/wav.c)
    target_link_libraries(accordomi-analyze PRIVATE accordomi)
    target_compile_options(accordomi-analyze PRIVATE -Wall -Wextra -Wpedantic -Werror)
    enable_testing()
    add_executable(accordomi-tests tests/test_engine.c)
    target_link_libraries(accordomi-tests PRIVATE accordomi)
    target_compile_options(accordomi-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME engine COMMAND accordomi-tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:accordomi-analyze>)
    endif()
endif()
